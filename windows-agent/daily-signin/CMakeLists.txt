cmake_minimum_required(VERSION 3.20)
project(RoomGoblinCredentialProvider LANGUAGES CXX)
add_library(RoomGoblinCredentialProvider SHARED provider.cpp provider.def)
target_compile_features(RoomGoblinCredentialProvider PRIVATE cxx_std_17)
target_compile_definitions(RoomGoblinCredentialProvider PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(RoomGoblinCredentialProvider PRIVATE ole32 uuid advapi32 secur32 credui shell32)
set_property(TARGET RoomGoblinCredentialProvider PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(RoomGoblinProviderTest provider-test.cpp)
target_compile_definitions(RoomGoblinProviderTest PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(RoomGoblinProviderTest PRIVATE ole32 uuid)
set_property(TARGET RoomGoblinProviderTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
